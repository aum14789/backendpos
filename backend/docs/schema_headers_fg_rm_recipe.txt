====================================================================================================
SunPOS Database Schema & Column Headers: 200 FG Items, Raw Materials (RM), Recipes, and Combo Sets
วันที่สร้าง: 2026-09-05 | เวอร์ชัน: 1.0 (Production Master Reference)
====================================================================================================

1. ตาราง MENU_CATEGORIES (หมวดหมู่อาหาร 8 หมวด)
   - id            : VARCHAR(36) PRIMARY KEY  (รหัสหมวดหมู่ เช่น cat-main, cat-shabu, cat-combo)
   - branch_id     : VARCHAR(36) NOT NULL     (รหัสสาขา)
   - name          : VARCHAR(100) NOT NULL    (ชื่อหมวดหมู่ เช่น อาหารจานเดียว, ชาบู, ซีฟู้ด, คอมโบ)
   - description   : TEXT                     (คำอธิบายหมวดหมู่)
   - sort_order    : INT DEFAULT 0            (ลำดับการแสดงผลบน POS)
   - is_active     : BOOLEAN DEFAULT TRUE     (สถานะเปิดใช้งาน)

2. ตาราง MENU_ITEMS (สินค้าสำเร็จรูป FG 200 รายการ)
   - id               : VARCHAR(36) PRIMARY KEY  (รหัสสินค้า FG เช่น fg-001 ถึง fg-200)
   - branch_id        : VARCHAR(36) NOT NULL     (รหัสสาขา)
   - category_id      : VARCHAR(36) NOT NULL     (รหัสหมวดหมู่อ้างอิง menu_categories.id)
   - sku              : VARCHAR(100)             (รหัสสินค้า SKU เช่น MAIN-001, SEAWT-071, SET-191)
   - name             : VARCHAR(255) NOT NULL    (ชื่อรายการอาหาร)
   - description      : TEXT                     (คำอธิบายเมนูอาหาร)
   - base_price       : NUMERIC(15,4) NOT NULL   (ราคาขาย: รองรับทั้งมีราคา และ 0.00 บาทสำหรับของฟรี)
   - cost_price       : NUMERIC(15,4)            (ต้นทุนมาตรฐานของเมนู)
   - item_type        : VARCHAR(20) DEFAULT 'FG' (ประเภทสินค้า: 'FG' สินค้าสำเร็จรูป / 'RM' วัตถุดิบ)
   - special_type     : VARCHAR(20)              ('S' สำหรับชุดคอมโบเซ็ต / NULL เมนูปกติ)
   - is_vat_inclusive : BOOLEAN DEFAULT TRUE     (TRUE=ราคารวม VAT / FALSE=ราคาไม่รวม VAT มีป้าย +VAT บน POS)
   - vat_rate         : NUMERIC(5,2) DEFAULT 7.00(อัตราภาษีมูลค่าเพิ่ม 7%)
   - allow_decimal_qty: BOOLEAN DEFAULT FALSE    (TRUE=สั่งเป็นทศนิยมได้ เช่น ซีฟู้ดชั่ง กก. / FALSE=จำนวนเต็ม)
   - kitchen_station  : VARCHAR(50)              (สถานีส่งพิมพ์ครัว: MAIN_KITCHEN, BAR ฯลฯ)
   - availability     : VARCHAR(50)              (AVAILABLE=พร้อมขาย / SOLD_OUT=หมด)
   - is_active        : BOOLEAN DEFAULT TRUE     (เปิดใช้งานในระบบ)

3. ตาราง INVENTORY_ITEMS (วัตถุดิบดิบ RM สำหรับใช้ในสูตร)
   - id                : VARCHAR(36) PRIMARY KEY  (รหัสวัตถุดิบ RM เช่น rm-beef-wagyu, rm-prawn-river)
   - sku               : VARCHAR(100) UNIQUE      (รหัสวัตถุดิบ SKU เช่น RM-MEAT-001, RM-SEA-001)
   - name              : VARCHAR(255) NOT NULL    (ชื่อวัตถุดิบที่ใช้ในสูตร เช่น เนื้อวัวดิบ, กุ้งสด, ข้าวหอมมะลิ)
   - category_name     : VARCHAR(100)             (หมวดหมู่วัตถุดิบ: MEAT, SEAFOOD, PRODUCE, SAUCES, DAIRY)
   - unit              : VARCHAR(50) NOT NULL     (หน่วยนับสั่งซื้อ/สต็อกหลัก: kg, L, piece, pack)
   - base_unit         : VARCHAR(50) NOT NULL     (หน่วยฐานย่อยที่ใช้ตัดสต็อกในสูตร: g, ml, piece)
   - conversion_factor : NUMERIC(12,4)            (อัตราการแปลงหน่วย เช่น 1 kg = 1000 g)
   - min_stock_alert   : NUMERIC(12,4)            (จุดเตือนเมื่อสต็อกเหลือน้อย)
   - is_active         : BOOLEAN DEFAULT TRUE     (สถานะเปิดใช้งาน)

4. ตาราง RECIPES (สูตรอาหารผูกกับ FG 1:1 ครบ 200 รายการ)
   - id             : VARCHAR(36) PRIMARY KEY  (รหัสสูตรอาหาร เช่น rcp-fg-001)
   - menu_item_id   : VARCHAR(36) NOT NULL     (รหัสสินค้า FG ที่ผูกกับสูตรนี้)
   - name           : VARCHAR(255) NOT NULL    (ชื่อสูตรอาหาร เช่น สูตรมาตรฐาน: ข้าวผัดปูจัมโบ้)
   - version        : VARCHAR(50) DEFAULT 'v1.0'(เวอร์ชันของสูตร)
   - yield_quantity : NUMERIC(12,4) DEFAULT 1.0 (ผลผลิตที่ได้ต่อ 1 ครั้งที่ปรุง)
   - yield_unit     : VARCHAR(50)              (หน่วยผลผลิต เช่น portion, จาน, ชาม)
   - is_active      : BOOLEAN DEFAULT TRUE     (สถานะเปิดใช้งาน)

5. ตาราง RECIPE_INGREDIENTS (รายการวัตถุดิบ RM ที่ตัดสต็อกในสูตร)
   - id                : VARCHAR(36) PRIMARY KEY  (รหัสแถวส่วนผสมในสูตร)
   - recipe_id         : VARCHAR(36) NOT NULL     (รหัสสูตรอาหารหลักอ้างอิง recipes.id)
   - inventory_item_id : VARCHAR(36) NOT NULL     (รหัสวัตถุดิบ RM อ้างอิง inventory_items.id)
   - quantity          : NUMERIC(12,4) NOT NULL   (ปริมาณที่ใช้ตัดสต็อกต่อ 1 จาน เช่น 0.1500 kg)
   - unit              : VARCHAR(50) NOT NULL     (หน่วยนับตัดสต็อก เช่น kg, L, piece)
   - waste_percentage  : NUMERIC(5,2) DEFAULT 0.0 (เปอร์เซ็นต์สูญเสียจากการเตรียมวัตถุดิบ)

6. ตาราง COMBO_DEFINITIONS & COMBO_GROUPS & COMBO_CHOICES (ชุดคอมโบเซ็ต KFC-Style)
   - combo_definitions.id       : VARCHAR(36) PRIMARY KEY (รหัสชุดคอมโบ)
   - combo_definitions.menu_item_id : รหัสสินค้า FG ที่เป็นชุดเซ็ต (special_type = 'S')
   - combo_groups.id            : VARCHAR(36) PRIMARY KEY (กลุ่มตัวเลือก เช่น เลือกจานหลัก, เลือกน้ำ)
   - combo_groups.min_selection : INT (จำนวนต่ำสุดที่ต้องเลือก ปกติ = 1)
   - combo_groups.max_selection : INT (จำนวนสูงสุดที่เลือกได้)
   - combo_choices.id           : VARCHAR(36) PRIMARY KEY (ตัวเลือกย่อยในกลุ่ม)
   - combo_choices.menu_item_id : รหัสสินค้า FG เดี่ยวที่นำมารวมเป็นตัวเลือกในเซ็ต
   - combo_choices.is_free      : BOOLEAN (รวมในเซ็ตฟรี หรือมีราคาเพิ่ม)
   - combo_choices.surcharge    : NUMERIC(15,4) (ราคาบวกเพิ่ม Upcharge หากเลือกเมนูพรีเมียม)
